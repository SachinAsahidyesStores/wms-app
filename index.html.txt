      function utf8ToBase64(str) {
        const bytes = new TextEncoder().encode(str);
        let bin = "";
        for (let i = 0; i < bytes.length; i++) {
          bin += String.fromCharCode(bytes[i]);
        }
        return btoa(bin);
      }

      function base64ToUtf8(b64) {
        const clean = b64.replace(/\s/g, "");
        const bin = atob(clean);
        const bytes = new Uint8Array(bin.length);
        for (let i = 0; i < bin.length; i++) {
          bytes[i] = bin.charCodeAt(i);
        }
        return new TextDecoder().decode(bytes);
      }

            const meta = await latest.json();
            body.sha = meta.sha;
            res = await ghApiFetch(filePath, {
              method: "PUT",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify(body)
            });
          }
        }

        if (!res.ok) {
          throw new Error(`Cloud save failed (${res.status})`);
        }
        const saved = await res.json();
        return saved.content ? saved.content.sha : null;

      function openIdb() {
        return new Promise((resolve, reject) => {
          if (!window.indexedDB) {
            reject(new Error("IndexedDB unavailable"));
            return;
          }
          const req = indexedDB.open(IDB_NAME, 1);

            const db = e.target.result;
            if (!db.objectStoreNames.contains(IDB_STORE)) {
              db.createObjectStore(IDB_STORE);
            }
          };
          req.onsuccess = () => resolve(req.result);
          req.onerror = () => reject(req.error);
        });
      }

      function formatDisplayDate(isoStr) {

        } else {

        }
      }

      function getExpiryInfo(expDateStr) {
        if (!expDateStr || expDateStr === "-") return { status: "NONE", daysLeft: Infinity, badgeHtml: "" };
        const today = new Date(todayISO() + "T00:00:00");
        const exp = new Date(expDateStr + "T00:00:00");
        if (Number.isNaN(exp.getTime())) return { status: "NONE", daysLeft: Infinity, badgeHtml: "" };
        const diffDays = Math.floor((exp.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
        if (diffDays < 0) {
          return { status: "EXPIRED", daysLeft: diffDays, badgeHtml: `<span class="badge badge-expired">EXPIRED (${Math.abs(diffDays)}d ago)</span>` };
        }
        if (diffDays <= 90) {
          return { status: "WARN", daysLeft: diffDays, badgeHtml: `<span class="badge badge-exp-warn">EXP IN ${diffDays}d</span>` };
        }
        return { status: "VALID", daysLeft: diffDays, badgeHtml: `<span class="badge badge-empty">VALID</span>` };
      }

      function generateId(prefix) {
        return `${prefix}-${Date.now().toString(36).toUpperCase()}-${Math.random().toString(36).substring(2, 6).toUpperCase()}`;
      }

      function escapeHtml(val) {
        if (val === null || val === undefined) return "";
        return String(val)
          .replace(/&/g, "&amp;")
          .replace(/</g, "&lt;")
          .replace(/>/g, "&gt;")
          .replace(/"/g, "&quot;")
          .replace(/'/g, "&#39;");
      }

      function parseRackParts(code) {
        const m = String(code).match(/^(\d{2})-([A-D])-(\d{2})$/i);
        if (m) return { line: m[1], level: m[2].toUpperCase(), bay: m[3] };
        return { line: "OTHER", level: "OTHER", bay: code };
      }

      function getSearchMatchedRacks(query) {
        const q = query.trim().toLowerCase();
        if (!q) return new Set();
        const matches = new Set();

        state.racks.forEach(code => {
          if (code.toLowerCase().includes(q)) matches.add(code);
        });

        state.inventory.forEach(item => {
          if (item.qty <= 0) return;
          const hay = `${item.rackCode} ${item.chemName} ${item.lotNo} ${item.invoiceNo} ${item.packingListNo} ${item.grnNo} ${item.mfgDate} ${item.expDate} ${item.pkgType} ${item.vendor}`.toLowerCase();
          if (hay.includes(q)) matches.add(item.rackCode);
        });

        return matches;
      }

      function renderRackGrid() {
        const container = document.getElementById("rackGridContainer");
        const occSet = getOccupiedRacksSet();
        const matchSet = getSearchMatchedRacks(ui.searchQuery);

        const clearBtn = document.getElementById("btnClearGlobalSearch");
        const matchPill = document.getElementById("pillSearchMatches");

        if (ui.searchQuery.trim()) {
          clearBtn.style.display = "inline-block";
          matchPill.style.display = "inline-block";
          matchPill.textContent = `Search Matches (${matchSet.size})`;
        } else {
          clearBtn.style.display = "none";
          matchPill.style.display = "none";
          if (ui.filterStatus === "MATCH") {
            ui.filterStatus = "ALL";
            document.querySelectorAll("#mapStatusPills .filter-pill").forEach(p => {
              p.classList.toggle("active", p.getAttribute("data-status") === "ALL");
            });
          }
        }

        const grouped = new Map();
        state.racks.forEach(code => {
          const isOcc = occSet.has(code);
          const isMatch = matchSet.has(code);
          const parts = parseRackParts(code);
          const lots = isOcc ? getLotsInRack(code) : [];

          let hasExpWarn = false;
          let hasExpired = false;
          lots.forEach(l => {
            const e = getExpiryInfo(l.expDate);
            if (e.status === "WARN") hasExpWarn = true;
            if (e.status === "EXPIRED") hasExpired = true;
          });

          if (ui.filterRackLine !== "ALL" && parts.line !== ui.filterRackLine) return;
          if (ui.filterShelfLevel !== "ALL" && parts.level !== ui.filterShelfLevel) return;
          if (ui.filterStatus === "EMPTY" && isOcc) return;
          if (ui.filterStatus === "OCCUPIED" && !isOcc) return;
          if (ui.filterStatus === "EXPIRING" && !hasExpWarn && !hasExpired) return;
          if (ui.filterStatus === "MATCH" && !isMatch) return;

          if (!grouped.has(parts.line)) grouped.set(parts.line, new Map());
          const lvlMap = grouped.get(parts.line);
          if (!lvlMap.has(parts.level)) lvlMap.set(parts.level, []);
          lvlMap.get(parts.level).push({ code, isOcc, isMatch, lots, hasExpWarn, hasExpired });
        });

        if (grouped.size === 0) {
          container.innerHTML = `<div class="empty-state">No rack locations match the selected filter criteria.</div>`;
          return;
        }

        const sectionHtml = [];
        grouped.forEach((lvlMap, lineKey) => {
          const lineLabel = lineKey === "OTHER" ? "Additional / Custom Locations" : `Rack Line ${lineKey}`;
          let totalInSection = 0;
          let occInSection = 0;
          lvlMap.forEach(arr => {
            totalInSection += arr.length;
            arr.forEach(x => { if (x.isOcc) occInSection++; });
          });

          const levelRowsHtml = [];
          lvlMap.forEach((tiles, lvlKey) => {
            const tagLabel = lvlKey === "OTHER" ? "CUSTOM" : `LVL ${lvlKey}`;
            const tilesHtml = tiles.map(t => {
              const cls = ["rack-tile"];
              if (t.isOcc) cls.push("occupied");
              if (t.hasExpWarn) cls.push("exp-warn");
              if (t.hasExpired) cls.push("expired");
              if (t.isMatch) cls.push("match");
              if (ui.selectedRackCode === t.code) cls.push("selected");

              let chemLine = "";
              let qtyPkgLine = "Empty";

              if (t.lots.length === 1) {
                const l = t.lots[0];
                chemLine = l.chemName;
                qtyPkgLine = l.pkgCount > 0 ? `${formatQty(l.qty)} ${l.unit} • ${l.pkgCount} ${l.pkgType}` : `${formatQty(l.qty)} ${l.unit}`;
              } else if (t.lots.length > 1) {
                chemLine = `${t.lots.length} Chemicals/Lots`;
                const totPkg = t.lots.reduce((a, b) => a + Number(b.pkgCount || 0), 0);
                qtyPkgLine = totPkg > 0 ? `${totPkg} Pkgs total` : `${t.lots.length} batches`;
              }

              const tip = t.lots.length
                ? `${t.code}: ` + t.lots.map(l => `${l.chemName} (Lot: ${l.lotNo || "-"}) — ${formatQtyWithPkg(l.qty, l.unit, l.pkgCount, l.pkgType)}`).join(" | ")
                : `${t.code}: Empty`;

              return `
                <div class="${cls.join(" ")}" data-rack-code="${escapeHtml(t.code)}" title="${escapeHtml(tip)}">
                  <div class="rack-tile-code">${escapeHtml(t.code)}</div>
                  ${chemLine ? `<div class="rack-tile-chem">${escapeHtml(chemLine)}</div>` : ""}
                  <div class="rack-tile-meta">${escapeHtml(qtyPkgLine)}</div>
                </div>
              `;
            }).join("");

            levelRowsHtml.push(`
              <div class="rack-level-row">
                <div class="rack-level-tag">${escapeHtml(tagLabel)}</div>
                <div class="rack-bay-grid">${tilesHtml}</div>
              </div>
            `);
          });

          sectionHtml.push(`
            <div class="rack-line-section">
              <div class="rack-line-header">
                <span class="rack-line-title">${escapeHtml(lineLabel)}</span>
                <span style="font-size:11.5px; color:var(--text-muted); font-family:var(--font-mono);">
                  Occupied: <strong style="color:#2563eb;">${occInSection}</strong> / ${totalInSection}
                </span>
              </div>
              ${levelRowsHtml.join("")}
            </div>
          `);
        });

        container.innerHTML = sectionHtml.join("");
      }

      function saveInlineRackRename() {
        const oldCode = ui.selectedRackCode;
        const newCode = document.getElementById("inspRenameInput").value.trim().toUpperCase();
        if (!newCode) {
          showToast("Rack location code cannot be empty.", "warning");
          return;
        }
        if (newCode === oldCode) return;

        if (state.racks.some(r => r.toUpperCase() === newCode)) {
          showToast(`Rack location "${newCode}" already exists.`, "warning");
          return;
        }

        const idx = state.racks.indexOf(oldCode);
        if (idx !== -1) state.racks[idx] = newCode;
        state.inventory.forEach(item => {
          if (item.rackCode === oldCode) item.rackCode = newCode;
        });
        ui.selectedRackCode = newCode;

        persistState();
        refreshAllViews();
        showToast(`Renamed rack location ${oldCode} ➔ ${newCode}`, "success");
      }
